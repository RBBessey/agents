# Python program to calculate the first 10,000 terms of the series 1 - 1/3 + 1/5 - 1/7 + ...
# and then multiply the result by 4.

def calculate_series(n_terms):
    total_sum = 0
    for i in range(n_terms):
        # Alternate adding and subtracting from 1,1/3,1/5,1/7,...
        term = 1 / (2 * i + 1)  # Denominator pattern
        if i % 2 == 0:  # Even index: add
            total_sum += term
        else:           # Odd index: subtract
            total_sum -= term
    return total_sum * 4  # Multiply the final sum by 4

# Calculate the first 10,000 terms
result = calculate_series(10000)
print(result)
